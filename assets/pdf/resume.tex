\section*{Projects}

\textbf{pdft: Trainable Unitary Bases for Image Compression and Inpainting} \hfill {\em Dec. 2025-Present} \\
\textcolor{textgray}{JAX library of trainable sparse bases (QFT, entangled QFT, TEBD, MERA, DCT-IV) optimized on the unitary manifold so the transform stays exactly invertible; the code behind arXiv:2608.00053 and arXiv:2609.17298. Supersedes the Julia original ParametricDFT.jl. \href{https://github.com/zazabap/pdft}{lib} \href{https://github.com/zazabap/pdft-benchmarks}{bench}}
\textit{\color{accentgreen}Python, JAX, Julia, Manopt.jl}

\textbf{problem-reductions: Extensible Framework for Computational Complexity} \hfill {\em Jan. 2026-Apr. 2026} \\
\textcolor{textgray}{Rust library and \texttt{pred} CLI cataloguing 100+ NP-hard problems and 200+ reductions, routing any problem to a registered solver through the reduction graph; developed with a human--agent engineering pipeline (arXiv:2604.11535). \href{https://github.com/CodingThrust/problem-reductions/}{lib}}
\textit{\color{accentgreen}Rust, Claude Code}

\textbf{LogosQ: Quantum Computing Library in Rust} \hfill {\em July 2025-Present} \\
\textcolor{textgray}{Type-safe simulator with gates, circuits, and optimized direct-gate and tensor-contraction backends (arXiv:2512.23183). \href{https://github.com/zazabap/LogosQ}{lib} \href{https://logosqbook.vercel.app/}{doc}}
\textit{\color{accentgreen}Rust, TypeScript, Docker}

\textbf{ManifoldsGPU.jl: GPU-Accelerated Riemannian Optimization} \hfill {\em Jan. 2026-Present} \\
\textcolor{textgray}{CUDA acceleration for the JuliaManifolds ecosystem: batched kernels for manifold operations and retractions, about $20\times$ faster \texttt{exp!} on \texttt{PowerManifold(Stiefel(32,16), 2048)} and $15\times$ for batched-SVD polar retraction. \href{https://github.com/JuliaManifolds/ManifoldsGPU.jl}{lib}}
\textit{\color{accentgreen}Julia, CUDA.jl, Manifolds.jl}
